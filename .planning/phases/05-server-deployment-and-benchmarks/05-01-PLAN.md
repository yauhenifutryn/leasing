install_apt_packages() {
  log "Installing apt packages"
  sudo apt update
  sudo apt install -y git curl unzip python3 python3-venv python3-pip \
    docker.io docker-compose-plugin ubuntu-drivers-common jq
  sudo usermod -aG docker "$USER" || true
}

**Step 2: `check_nvidia_driver()`** -- Per D-04:
```bash
check_nvidia_driver() {
  if nvidia-smi &>/dev/null; then
    log "NVIDIA driver OK: $(nvidia-smi --query-gpu=name --format=csv,noheader)"
    return 0
  fi
  log "NVIDIA driver not found -- installing via ubuntu-drivers"
  sudo apt install -y ubuntu-drivers-common
  sudo ubuntu-drivers install
  log "Driver installed. REBOOT REQUIRED."
  log "After reboot, re-run: $0"
  exit 1
}
```

**Step 3: `clone_repo()`** -- Clone if not present, pull if already cloned:
```bash
clone_repo() {
  if [ -d "$WORKSPACE/leasing/.git" ]; then
    log "Repo exists; pulling latest"
    cd "$WORKSPACE/leasing" && git pull && cd -
  else
    log "Cloning repo"
    mkdir -p "$WORKSPACE"
    git clone --branch "$REPO_BRANCH" "$REPO_URL" "$WORKSPACE/leasing"
  fi
}
```

**Step 4: `ensure_venv()`** -- Reuse pattern from setup_vast_voice.sh:
```bash
ensure_venv() {
  local target="$1"
  local req_file="$2"
  if [ ! -d "$target" ]; then
    log "Creating venv: $target"
    python3 -m venv "$target"
  fi
  "$target/bin/pip" install --upgrade pip wheel
  "$target/bin/pip" install -r "$req_file"
}
```
